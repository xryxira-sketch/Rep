local function New(class, props, parent)
	local obj = Instance.new(class)
	for k, v in pairs(props or {}) do
		obj[k] = v
	end
	obj.Parent = parent
	return obj
end

local function Corner(obj, r)
	local c = Instance.new("UICorner")
	c.CornerRadius = UDim.new(0, r)
	c.Parent = obj
	return c
end

local function Stroke(obj, color, tr, th)
	local s = Instance.new("UIStroke")
	s.Color = color
	s.Transparency = tr or 0
	s.Thickness = th or 1
	s.Parent = obj
	return s
end

local function Tween(obj, props, info)
	local t = TweenService:Create(obj, info or Fast, props)
	t:Play()
	return t
end

local Character, Humanoid, Root

local function UpdateCharacter()
	local char = Player.Character
	if not char then
		Character, Humanoid, Root = nil, nil, nil
		return
	end
	Character = char
	Humanoid = char:FindFirstChildOfClass("Humanoid")
	Root = char:FindFirstChild("HumanoidRootPart")
end

UpdateCharacter()
Player.CharacterAdded:Connect(function()
	task.wait(0.3)
	UpdateCharacter()
	if Settings.WalkSpeed and Humanoid then
		Humanoid.WalkSpeed = Settings.WalkSpeed
	end
end)

	while Gui.Parent do
		if not (Root and Root.Parent) then

local FloatConnection, LiftConnection, NoclipConnection, TPWalkConnection, JumpConnection
local LiftTimerConnection, StreamKeepAliveConnection
local FloatBaseY
local LiftStartTime = 0
local StreamKeepAliveEnabled = true

local Checkpoint1, Checkpoint2
local TeleportIndex = 1

local function Disconnect(c)
	if c then c:Disconnect() end
	return nil
end

local function StopFloat()
	FloatConnection = Disconnect(FloatConnection)
	FloatBaseY = nil
	if Root and Root.Parent then
		local v = Root.AssemblyLinearVelocity
		Root.AssemblyLinearVelocity = Vector3.new(v.X, 0, v.Z)
	end
end

local function StartFloat()
	StopFloat()
	if not Root or not Root.Parent then return end
	FloatBaseY = Root.Position.Y + Settings.FloatHeight
	FloatConnection = RunService.Heartbeat:Connect(function()
		if not Settings.Float or not Root or not Root.Parent then return end
		local p = Root.Position
		local v = Root.AssemblyLinearVelocity
		Root.AssemblyLinearVelocity = Vector3.new(v.X, 0, v.Z)
		Root.CFrame = CFrame.new(p.X, FloatBaseY, p.Z) * CFrame.Angles(
			math.rad(Root.Orientation.X),
			math.rad(Root.Orientation.Y),
			math.rad(Root.Orientation.Z)
		)
	end)
end

local function SetFloat(on)
	Settings.Float = on
	if on then StartFloat() else StopFloat() end
end

local function StopStreamKeepAlive()
	if typeof(StreamKeepAliveConnection) == "thread" then
		pcall(task.cancel, StreamKeepAliveConnection)
	end
	StreamKeepAliveConnection = nil
end

local function StartStreamKeepAlive()
	StopStreamKeepAlive()
	StreamKeepAliveConnection = task.spawn(function()
		while Settings.Lift and StreamKeepAliveEnabled and Gui.Parent do
			pcall(function()
				if Root and Root.Parent then
					Player:RequestStreamAroundAsync(Root.Position, 2)
				end
				if Checkpoint1 then
					Player:RequestStreamAroundAsync(Checkpoint1.Position, 2)
				end
			end)
			task.wait(2)
		end
		StreamKeepAliveConnection = nil
	end)
end

local function StopLiftTimer()
	LiftTimerConnection = Disconnect(LiftTimerConnection)
	LiftStartTime = 0
end

local function StopLift()
	LiftConnection = Disconnect(LiftConnection)
	StopLiftTimer()
	StopStreamKeepAlive()
	if Root and Root.Parent then
		local v = Root.AssemblyLinearVelocity
		Root.AssemblyLinearVelocity = Vector3.new(v.X, 0, v.Z)
	end
end

local UnitMult = {
	Nanoseconds = 1e-9, Microseconds = 1e-6, Milliseconds = 0.001,
	Seconds = 1, Minutes = 60, Hours = 3600, Days = 86400,
	Weeks = 604800, Months = 2592000, Years = 31536000,
	Decades = 315360000, Centuries = 3153600000,
}
local UnitList = {
	"Nanoseconds","Microseconds","Milliseconds","Seconds","Minutes",
	"Hours","Days","Weeks","Months","Years","Decades","Centuries"
}

local status -- filled later

local function TriggerLiftTimeout()
	if not Root or not Root.Parent then
		LiftStartTime = os.clock()
		return
	end
	local target = Checkpoint1 or Checkpoint2
	local label = Checkpoint1 and "CP1" or (Checkpoint2 and "CP2" or nil)
	if target then
		Root.CFrame = target
		local v = Root.AssemblyLinearVelocity
		Root.AssemblyLinearVelocity = Vector3.new(v.X, 0, v.Z)
		if status then status.Text = "Lift timer → " .. label .. " (lift still on)" end
	else
		if status then status.Text = "Lift timer — no checkpoint saved" end
	end
	LiftStartTime = os.clock()
end

local function StartLiftTimer()
	StopLiftTimer()
	local amount = tonumber(Settings.LiftTimerValue) or 0
	if amount <= 0 then return end
	local mult = UnitMult[Settings.LiftTimerUnit] or 60
	local limit = math.max(amount * mult, 0.05)
	LiftStartTime = os.clock()
	LiftTimerConnection = RunService.Heartbeat:Connect(function()
		if not Settings.Lift then
			StopLiftTimer()
			return
		end
		if (os.clock() - LiftStartTime) >= limit then
			TriggerLiftTimeout()
		end
	end)
end

local function StartLift()
	StopLift()
	LiftConnection = RunService.Heartbeat:Connect(function()
		if not Settings.Lift or not Root or not Root.Parent then return end
		local v = Root.AssemblyLinearVelocity
		Root.AssemblyLinearVelocity = Vector3.new(v.X, Settings.LiftSpeed * 25, v.Z)
	end)
	StartLiftTimer()
	StartStreamKeepAlive()
end

local function SetLift(on)
	Settings.Lift = on
	if on then StartLift() else StopLift() end
end

local function StopNoclip()
	NoclipConnection = Disconnect(NoclipConnection)
	if Character then
		for _, o in ipairs(Character:GetDescendants()) do
			if o:IsA("BasePart") then o.CanCollide = true end
		end
	end
end

local function StartNoclip()
	StopNoclip()
	NoclipConnection = RunService.Stepped:Connect(function()
		if not Settings.Noclip or not Character then return end
		for _, o in ipairs(Character:GetDescendants()) do
			if o:IsA("BasePart") then o.CanCollide = false end
		end
	end)
end

local function SetNoclip(on)
	Settings.Noclip = on
	if on then StartNoclip() else StopNoclip() end
end

local function StopTPWalk()
	TPWalkConnection = Disconnect(TPWalkConnection)
end

local function StartTPWalk()
	StopTPWalk()
	TPWalkConnection = RunService.Heartbeat:Connect(function()
		if not Settings.TPWalk or not Humanoid or not Root then return end
		local d = Humanoid.MoveDirection
		if d.Magnitude > 0 then
			Root.CFrame = Root.CFrame + (d * Settings.TPWalkSpeed)
		end
	end)
end

local function SetTPWalk(on)
	Settings.TPWalk = on
	if on then StartTPWalk() else StopTPWalk() end
end

local function StopInfiniteJump()
	JumpConnection = Disconnect(JumpConnection)
end

local function StartInfiniteJump()
	StopInfiniteJump()
	JumpConnection = UserInputService.JumpRequest:Connect(function()
		if Settings.InfiniteJump and Humanoid and Humanoid.Parent then
			Humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
		end
	end)
end

local function SetInfiniteJump(on)
	Settings.InfiniteJump = on
	if on then StartInfiniteJump() else StopInfiniteJump() end
end

local function SetWalkSpeed(v)
	Settings.WalkSpeed = v
	if Humanoid and Humanoid.Parent then
		Humanoid.WalkSpeed = v
	end
end

local function SaveCheckpoint(n)
	if not Root or not Root.Parent then return false end
	if n == 1 then Checkpoint1 = Root.CFrame else Checkpoint2 = Root.CFrame end
	return true
end

local function TeleportCheckpoint()
	if not Root or not Root.Parent then return false end
	if TeleportIndex == 1 then
		if Checkpoint1 then Root.CFrame = Checkpoint1; TeleportIndex = 2; return true
		elseif Checkpoint2 then Root.CFrame = Checkpoint2; TeleportIndex = 1; return true end
	else
		if Checkpoint2 then Root.CFrame = Checkpoint2; TeleportIndex = 1; return true
		elseif Checkpoint1 then Root.CFrame = Checkpoint1; TeleportIndex = 2; return true end
	end
	return false
end

local Open = false

local function OpenMain()
	Open = true
	Main.Visible = true
	Main.Size = UDim2.fromOffset(420, 260)
	Tween(Main, {Size = UDim2.fromOffset(480, 300)}, Spring)
	Tween(CircleImg, {ImageColor3 = Theme.Green}, Fast)
end

local function CloseMain()
	Open = false
	Tween(Main, {Size = UDim2.fromOffset(420, 260)}, Fast)
	Tween(CircleImg, {ImageColor3 = Theme.White}, Fast)
	task.delay(0.2, function()
		if not Open then Main.Visible = false end
	end)
end

Circle.Activated:Connect(function()
	if Open then CloseMain() else OpenMain() end
end)
CloseBtn.Activated:Connect(CloseMain)

-- Drag main
do
	local dragging, start, startPos
	Header.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			start = input.Position
			startPos = Main.Position
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
			local d = input.Position - start
			Main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
		end
	end)
	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = false
		end
	end)
end

-- Drag orb
do
	local dragging, moved, start, startPos
	Circle.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			moved = false
			start = input.Position
			startPos = Circle.Position
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if not dragging then return end
		if input.UserInputType ~= Enum.UserInputType.MouseMovement and input.UserInputType ~= Enum.UserInputType.Touch then return end
		local d = input.Position - start
		if d.Magnitude > 8 then moved = true end
		Circle.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
	end)
	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = false
		end
	end)
end

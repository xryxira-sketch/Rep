local function New(class, props, parent)
	local obj = Instance.new(class)

	for property, value in pairs(props or {}) do
		obj[property] = value
	end

	obj.Parent = parent
	return obj
end

local function Corner(obj, radius)
	local c = Instance.new("UICorner")
	c.CornerRadius = UDim.new(0, radius)
	c.Parent = obj
	return c
end

local function Stroke(obj, color, transparency, thickness)
	local s = Instance.new("UIStroke")
	s.Color = color
	s.Transparency = transparency or 0
	s.Thickness = thickness or 1
	s.Parent = obj
	return s
end

local function Gradient(obj, colors, rotation)
	local keys = {}

	for i, color in ipairs(colors) do
		keys[#keys + 1] = ColorSequenceKeypoint.new(
			(i - 1) / (#colors - 1),
			color
		)
	end

	local g = Instance.new("UIGradient")
	g.Color = ColorSequence.new(keys)
	g.Rotation = rotation or 0
	g.Parent = obj

	return g
end

local function Tween(obj, properties, info)
	local t = TweenService:Create(
		obj,
		info or Fast,
		properties
	)

	t:Play()
	return t
end

--========================================================--
-- FLOAT
--========================================================--

local FloatBaseY

local function StopFloat()
	FloatConnection = Disconnect(FloatConnection)
	FloatBaseY = nil

	-- Remove artificial vertical velocity rather than
	-- suddenly forcing the character downward/upward.
	if Root and Root.Parent then
		local velocity = Root.AssemblyLinearVelocity

		Root.AssemblyLinearVelocity = Vector3.new(
			velocity.X,
			0,
			velocity.Z
		)
	end
end

local function StartFloat()
	StopFloat()

	if not Root or not Root.Parent then
		return
	end

	FloatBaseY = Root.Position.Y + Settings.FloatHeight

	FloatConnection = RunService.Heartbeat:Connect(function()
		if not Root or not Root.Parent then
			return
		end

		if not Settings.Float then
			return
		end

		local p = Root.Position

		-- Preserve horizontal movement.
		-- Only control the vertical coordinate.
		Root.AssemblyLinearVelocity = Vector3.new(
			Root.AssemblyLinearVelocity.X,
			0,
			Root.AssemblyLinearVelocity.Z
		)

		Root.CFrame = CFrame.new(
			p.X,
			FloatBaseY,
			p.Z
		) * CFrame.Angles(
			Root.Orientation.X * math.pi / 180,
			Root.Orientation.Y * math.pi / 180,
			Root.Orientation.Z * math.pi / 180
		)
	end)
end

local function SetFloat(value)
	Settings.Float = value

	if value then
		StartFloat()
	else
		StopFloat()
	end
end

--========================================================--
-- INFINITE LIFT
--========================================================--

local function StopLift()
	LiftConnection = Disconnect(LiftConnection)

	if Root and Root.Parent then
		local velocity = Root.AssemblyLinearVelocity

		Root.AssemblyLinearVelocity = Vector3.new(
			velocity.X,
			0,
			velocity.Z
		)
	end
end

local function StartLift()
	StopLift()

	LiftConnection = RunService.Heartbeat:Connect(function()
		if not Settings.Lift then
			return
		end

		if not Root or not Root.Parent then
			return
		end

		local velocity = Root.AssemblyLinearVelocity

		Root.AssemblyLinearVelocity = Vector3.new(
			velocity.X,
			Settings.LiftSpeed * 25,
			velocity.Z
		)
	end)
end

local function SetLift(value)
	Settings.Lift = value

	if value then
		StartLift()
	else
		StopLift()
	end
end

--========================================================--
-- NOCLIP
--========================================================--

local function RestoreCollisions()
	if not Character then
		return
	end

	for _, object in ipairs(Character:GetDescendants()) do
		if object:IsA("BasePart") then
			object.CanCollide = true
		end
	end
end

local function StopNoclip()
	NoclipConnection = Disconnect(NoclipConnection)
	RestoreCollisions()
end

local function StartNoclip()
	StopNoclip()

	NoclipConnection = RunService.Stepped:Connect(function()
		if not Settings.Noclip then
			return
		end

		if not Character then
			return
		end

		for _, object in ipairs(Character:GetDescendants()) do
			if object:IsA("BasePart") then
				object.CanCollide = false
			end
		end
	end)
end

local function SetNoclip(value)
	Settings.Noclip = value

	if value then
		StartNoclip()
	else
		StopNoclip()
	end
end

--========================================================--
-- TPWALK
--========================================================--

local function StopTPWalk()
	TPWalkConnection = Disconnect(TPWalkConnection)
end

local function StartTPWalk()
	StopTPWalk()

	TPWalkConnection = RunService.Heartbeat:Connect(function()
		if not Settings.TPWalk then
			return
		end

		if not Humanoid or not Root then
			return
		end

		local direction = Humanoid.MoveDirection

		if direction.Magnitude > 0 then
			Root.CFrame =
				Root.CFrame +
				(direction * Settings.TPWalkSpeed)
		end
	end)
end

local function SetTPWalk(value)
	Settings.TPWalk = value

	if value then
		StartTPWalk()
	else
		StopTPWalk()
	end
end

--========================================================--
-- CHECKPOINT SYSTEM
--========================================================--

local Checkpoint1
local Checkpoint2
local TeleportIndex = 1

local function SaveCheckpoint(number)
	if not Root or not Root.Parent then
		return false
	end

	if number == 1 then
		Checkpoint1 = Root.CFrame
	elseif number == 2 then
		Checkpoint2 = Root.CFrame
	end

	return true
end

local function ResetCheckpoints()
	Checkpoint1 = nil
	Checkpoint2 = nil
	TeleportIndex = 1
end

local function TeleportCheckpoint()
	if not Root or not Root.Parent then
		return false
	end

	-- Normal cycle:
	-- CP1 → CP2 → CP1 → CP2
	if TeleportIndex == 1 then

		if Checkpoint1 then
			Root.CFrame = Checkpoint1
			TeleportIndex = 2
			return true
		elseif Checkpoint2 then
			Root.CFrame = Checkpoint2
			TeleportIndex = 1
			return true
		end

	else

		if Checkpoint2 then
			Root.CFrame = Checkpoint2
			TeleportIndex = 1
			return true
		elseif Checkpoint1 then
			Root.CFrame = Checkpoint1
			TeleportIndex = 2
			return true
		end

	end

	return false
end

local CurrentPage

local function ShowPage(name)
	for pageName, page in pairs(Pages) do
		local selected = pageName == name

		page.Visible = selected

		local data = PageButtons[pageName]

		if selected then
			Tween(
				data.Button,
				{BackgroundColor3 = Color3.fromRGB(38, 27, 64)},
				Fast
			)

			Tween(
				data.Icon,
				{TextColor3 = Theme.Violet},
				Fast
			)

			Tween(
				data.Label,
				{TextColor3 = Theme.White},
				Fast
			)
		else
			Tween(
				data.Button,
				{BackgroundColor3 = Theme.Card},
				Fast
			)

			Tween(
				data.Icon,
				{TextColor3 = Theme.Gray},
				Fast
			)

			Tween(
				data.Label,
				{TextColor3 = Theme.Gray},
				Fast
			)
		end
	end

	CurrentPage = name
end

for name, data in pairs(PageButtons) do
	data.Button.Activated:Connect(function()
		ShowPage(name)
	end)
end

	end)

	return {
		SetVisual = UpdateVisual

local function MakeDraggable(object, handle)
	local dragging = false
	local dragStart
	local startPosition
	local dragInput

	handle.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then

			dragging = true
			dragStart = input.Position
			startPosition = object.Position
			dragInput = input
		end
	end)

	handle.InputChanged:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseMovement
			or input.UserInputType == Enum.UserInputType.Touch then

			dragInput = input
		end
	end)

	UserInputService.InputChanged:Connect(function(input)
		if dragging and input == dragInput then

			local delta =
				input.Position - dragStart

			object.Position = UDim2.new(
				startPosition.X.Scale,
				startPosition.X.Offset + delta.X,

				startPosition.Y.Scale,
				startPosition.Y.Offset + delta.Y
			)
		end
	end)

	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then

			dragging = false
		end
	end)
end

local CircleDragging = false
local CircleMoved = false
local CircleDragStart
local CircleStartPosition

Circle.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1
		or input.UserInputType == Enum.UserInputType.Touch then

		CircleDragging = true
		CircleMoved = false

		CircleDragStart = input.Position
		CircleStartPosition = Circle.Position
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if not CircleDragging then
		return
	end

	if input.UserInputType ~= Enum.UserInputType.MouseMovement
		and input.UserInputType ~= Enum.UserInputType.Touch then
		return
	end

	local delta =
		input.Position - CircleDragStart

	if delta.Magnitude > 8 then
		CircleMoved = true
	end

	Circle.Position = UDim2.new(
		CircleStartPosition.X.Scale,
		CircleStartPosition.X.Offset + delta.X,

		CircleStartPosition.Y.Scale,
		CircleStartPosition.Y.Offset + delta.Y
	)
end)

UserInputService.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1
		or input.UserInputType == Enum.UserInputType.Touch then

		CircleDragging = false
	end
end)

-- XYLIX / XYRIX HIGH-ALTITUDE STREAM STABILITY MODULE
-- Added: targeted streaming keep-alive + pre-teleport prefetch
--========================================================--

local XylixStream = {
    Enabled = true,
    Interval = 3,
    Timeout = 2,
    PreTeleportLead = 4,
    LastRequest = 0,
}

local function XylixGetRoot()
    local c = Player.Character
    if not c then return nil end
    return c:FindFirstChild("HumanoidRootPart")
end

local function XylixRequest(position)
    if not position then return end
    local ok = pcall(function()
        Player:RequestStreamAroundAsync(position, XylixStream.Timeout)
    end)
    return ok
end

local function XylixPrefetchCurrentAndCheckpoint()
    if not XylixStream.Enabled then return end

    local root = XylixGetRoot()
    if not root then return end

    local now = os.clock()
    if now - XylixStream.LastRequest < XylixStream.Interval then
        return
    end
    XylixStream.LastRequest = now

    XylixRequest(root.Position)

    -- Prefetch the checkpoint that the lift timer is most likely
    -- to use next. This is intentionally one targeted request,
    -- rather than trying to load the entire map.
    local target = nil
    if TeleportIndex == 1 then
        target = Checkpoint1 or Checkpoint2
    else
        target = Checkpoint2 or Checkpoint1
    end

    if target then
        XylixRequest(target.Position)
    end
end

local function XylixPrefetchBeforeTeleport(target)
    if not target then return end
    XylixRequest(target.Position)
end

-- Lightweight background loop. It does not run every frame.
task.spawn(function()
    while Gui and Gui.Parent do
        if Settings and Settings.Lift then
            XylixPrefetchCurrentAndCheckpoint()
        end
        task.wait(0.5)
    end
end)

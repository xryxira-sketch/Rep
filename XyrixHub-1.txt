--========================================================--
-- HELPERS

local function New(class, props, parent)
	local obj = Instance.new(class)

	for property, value in pairs(props or {}) do
		obj[property] = value
	end

	obj.Parent = parent
	return obj
end

local function Corner(obj, radius)
	local c = Instance.new("UICorner")
	c.CornerRadius = UDim.new(0, radius)
	c.Parent = obj
	return c
end

local function Stroke(obj, color, transparency, thickness)
	local s = Instance.new("UIStroke")
	s.Color = color
	s.Transparency = transparency or 0
	s.Thickness = thickness or 1
	s.Parent = obj
	return s
end

local function Gradient(obj, colors, rotation)
	local keys = {}

	for i, color in ipairs(colors) do
		keys[#keys + 1] = ColorSequenceKeypoint.new(
			(i - 1) / (#colors - 1),
			color
		)
	end

	local g = Instance.new("UIGradient")
	g.Color = ColorSequence.new(keys)
	g.Rotation = rotation or 0
	g.Parent = obj

	return g
end

local function Tween(obj, properties, info)
	local t = TweenService:Create(
		obj,
		info or Fast,
		properties
	)

	t:Play()
	return t
end

-- FLOAT
--========================================================--

local FloatBaseY

local function StopFloat()
	FloatConnection = Disconnect(FloatConnection)
	FloatBaseY = nil

	-- Remove artificial vertical velocity rather than
	-- suddenly forcing the character downward/upward.
	if Root and Root.Parent then
		local velocity = Root.AssemblyLinearVelocity

		Root.AssemblyLinearVelocity = Vector3.new(
			velocity.X,
			0,
			velocity.Z
		)
	end
end

local function StartFloat()
	StopFloat()

	if not Root or not Root.Parent then
		return
	end

	FloatBaseY = Root.Position.Y + Settings.FloatHeight

	FloatConnection = RunService.Heartbeat:Connect(function()
		if not Root or not Root.Parent then
			return
		end

		if not Settings.Float then
			return
		end

		local p = Root.Position

		-- Preserve horizontal movement.
		-- Only control the vertical coordinate.
		Root.AssemblyLinearVelocity = Vector3.new(
			Root.AssemblyLinearVelocity.X,
			0,
			Root.AssemblyLinearVelocity.Z
		)

		Root.CFrame = CFrame.new(
			p.X,
			FloatBaseY,
			p.Z
		) * CFrame.Angles(
			Root.Orientation.X * math.pi / 180,
			Root.Orientation.Y * math.pi / 180,
			Root.Orientation.Z * math.pi / 180
		)
	end)
end

local function SetFloat(value)
	Settings.Float = value

	if value then
		StartFloat()
	else
		StopFloat()
	end
end

--========================================================--
-- INFINITE LIFT + AUTO TIMER
--========================================================--

local LiftStartTime = 0
local LiftTimerConnection
local StreamKeepAliveConnection
local StreamKeepAliveEnabled = true

local function StopLiftTimer()
	LiftTimerConnection = Disconnect(LiftTimerConnection)
	LiftStartTime = 0
end

local function StopStreamKeepAlive()
	if typeof(StreamKeepAliveConnection) == "thread" then
		task.cancel(StreamKeepAliveConnection)
	end
	StreamKeepAliveConnection = nil
end

local function StartStreamKeepAlive()
	StopStreamKeepAlive()

	-- RequestStreamAroundAsync keeps map chunks loaded while flying (reduces hitch/pause)
	StreamKeepAliveConnection = task.spawn(function()
		while Settings.Lift and StreamKeepAliveEnabled do
			pcall(function()
				if Root and Root.Parent then
					Player:RequestStreamAroundAsync(Root.Position, 5)
				end
				if Checkpoint1 then
					Player:RequestStreamAroundAsync(Checkpoint1.Position, 5)
				end
			end)
			task.wait(2)
		end
		StreamKeepAliveConnection = nil
	end)
end

local function StopLift()
	LiftConnection = Disconnect(LiftConnection)
	StopLiftTimer()
	StopStreamKeepAlive()

	if Root and Root.Parent then
		local velocity = Root.AssemblyLinearVelocity

		Root.AssemblyLinearVelocity = Vector3.new(
			velocity.X,
			0,
			velocity.Z
		)
	end
end

-- Unit multipliers to seconds
local TimerUnits = {
	{Name = "Nanoseconds", Multiplier = 1e-9},
	{Name = "Microseconds", Multiplier = 1e-6},
	{Name = "Milliseconds", Multiplier = 0.001},
	{Name = "Seconds", Multiplier = 1},
	{Name = "Minutes", Multiplier = 60},
	{Name = "Hours", Multiplier = 3600},
	{Name = "Days", Multiplier = 86400},
	{Name = "Weeks", Multiplier = 604800},
	{Name = "Months", Multiplier = 2592000}, -- ~30 days
	{Name = "Years", Multiplier = 31536000},
	{Name = "Decades", Multiplier = 315360000},
	{Name = "Centuries", Multiplier = 3153600000},
}

local function GetUnitMultiplier(unitName)
	for _, unit in ipairs(TimerUnits) do
		if unit.Name == unitName then
			return unit.Multiplier
		end
	end
	return 60 -- default minutes
end

local function GetNextUnit(current)
	for i, unit in ipairs(TimerUnits) do
		if unit.Name == current then
			local nextIndex = (i % #TimerUnits) + 1
			return TimerUnits[nextIndex].Name
		end
	end
	return "Minutes"
end

local function StartLiftTimer()
	StopLiftTimer()

	local amount = tonumber(Settings.LiftTimerValue) or 0
	if amount <= 0 then
		return -- 0 = disabled
	end

	local multiplier = GetUnitMultiplier(Settings.LiftTimerUnit or "Minutes")
	local limitSeconds = amount * multiplier

	-- Extremely small values still need a minimum practical tick
	if limitSeconds < 0.05 then
		limitSeconds = 0.05
	end

	LiftStartTime = os.clock()

	LiftTimerConnection = RunService.Heartbeat:Connect(function()
		if not Settings.Lift then
			StopLiftTimer()
			return
		end

		if (os.clock() - LiftStartTime) >= limitSeconds then
			TriggerLiftTimeout()
		end
	end)
end

local function TriggerLiftTimeout()
	-- Keep Infinite Lift ON. Teleport to checkpoint, then restart timer so it loops.
	if not Root or not Root.Parent then
		-- Still restart timer so the cycle continues when character is valid again
		LiftStartTime = os.clock()
		return
	end

	local target = Checkpoint1 or Checkpoint2
	local label = Checkpoint1 and "CP1" or (Checkpoint2 and "CP2" or nil)

	if target then
		Root.CFrame = target
		-- Clear residual vertical velocity so the teleport feels clean, then lift continues
		local v = Root.AssemblyLinearVelocity
		Root.AssemblyLinearVelocity = Vector3.new(v.X, 0, v.Z)

		if status then
			status.Text = "Lift timer → teleported to " .. label .. " (lift still on)."
		end
	else
		if status then
			status.Text = "Lift timer hit — no checkpoint saved."
		end
	end

	-- Restart the timer so the cycle repeats while Lift stays enabled
	LiftStartTime = os.clock()
end

local function StartLift()
	StopLift()

	LiftConnection = RunService.Heartbeat:Connect(function()
		if not Settings.Lift then
			return
		end

		if not Root or not Root.Parent then
			return
		end

		local velocity = Root.AssemblyLinearVelocity

		Root.AssemblyLinearVelocity = Vector3.new(
			velocity.X,
			Settings.LiftSpeed * 25,
			velocity.Z
		)
	end)

	StartLiftTimer()
	StartStreamKeepAlive()
end

local function SetLift(value)
	Settings.Lift = value

	if value then
		StartLift()
	else
		StopLift()
	end
end

local function RestoreCollisions()
	if not Character then
		return
	end

	for _, object in ipairs(Character:GetDescendants()) do
		if object:IsA("BasePart") then
			object.CanCollide = true
		end
	end
end

local function StopNoclip()
	NoclipConnection = Disconnect(NoclipConnection)
	RestoreCollisions()
end

local function StartNoclip()
	StopNoclip()

	NoclipConnection = RunService.Stepped:Connect(function()
		if not Settings.Noclip then
			return
		end

		if not Character then
			return
		end

		for _, object in ipairs(Character:GetDescendants()) do
			if object:IsA("BasePart") then
				object.CanCollide = false
			end
		end
	end)
end

local function SetNoclip(value)
	Settings.Noclip = value

	if value then
		StartNoclip()
	else
		StopNoclip()
	end
end

local function StopTPWalk()
	TPWalkConnection = Disconnect(TPWalkConnection)
end

local function StartTPWalk()
	StopTPWalk()

	TPWalkConnection = RunService.Heartbeat:Connect(function()
		if not Settings.TPWalk then
			return
		end

		if not Humanoid or not Root then
			return
		end

		local direction = Humanoid.MoveDirection

		if direction.Magnitude > 0 then
			Root.CFrame =
				Root.CFrame +
				(direction * Settings.TPWalkSpeed)
		end
	end)
end

local function SetTPWalk(value)
	Settings.TPWalk = value

	if value then
		StartTPWalk()
	else
		StopTPWalk()
	end
end

--========================================================--
-- CHECKPOINT SYSTEM
--========================================================--

local Checkpoint1
local Checkpoint2
local TeleportIndex = 1

local function SaveCheckpoint(number)
	if not Root or not Root.Parent then
		return false
	end

	if number == 1 then
		Checkpoint1 = Root.CFrame
	elseif number == 2 then
		Checkpoint2 = Root.CFrame
	end

	return true
end

local function ResetCheckpoints()
	Checkpoint1 = nil
	Checkpoint2 = nil
	TeleportIndex = 1
end

local function TeleportCheckpoint()
	if not Root or not Root.Parent then
		return false
	end

	-- Normal cycle:
	-- CP1 → CP2 → CP1 → CP2
	if TeleportIndex == 1 then

		if Checkpoint1 then
			Root.CFrame = Checkpoint1
			TeleportIndex = 2
			return true
		elseif Checkpoint2 then
			Root.CFrame = Checkpoint2
			TeleportIndex = 1
			return true
		end

	else

		if Checkpoint2 then
			Root.CFrame = Checkpoint2
			TeleportIndex = 1
			return true
		elseif Checkpoint1 then
			Root.CFrame = Checkpoint1
			TeleportIndex = 2
			return true
		end

	end

	return false
end

-- PAGE SWITCHING

local CurrentPage

local function ShowPage(name)
	for pageName, page in pairs(Pages) do
		local selected = pageName == name
		page.Visible = selected

		local data = PageButtons[pageName]

		if selected then
			Tween(data.Button, {
				BackgroundTransparency = 0,
				BackgroundColor3 = Color3.fromRGB(46, 32, 78),
			}, Fast)
			Tween(data.Fill, {BackgroundTransparency = 0.35}, Fast)
			Tween(data.Icon, {TextColor3 = Theme.Cyan}, Fast)
			Tween(data.Label, {TextColor3 = Theme.White}, Fast)
			Tween(data.Accent, {
				BackgroundTransparency = 0,
				BackgroundColor3 = Theme.Cyan,
			}, Fast)
		else
			Tween(data.Button, {
				BackgroundTransparency = 1,
				BackgroundColor3 = Color3.fromRGB(38, 28, 64),
			}, Fast)
			Tween(data.Fill, {BackgroundTransparency = 1}, Fast)
			Tween(data.Icon, {TextColor3 = Theme.Gray}, Fast)
			Tween(data.Label, {TextColor3 = Theme.Gray}, Fast)
			Tween(data.Accent, {
				BackgroundTransparency = 1,
				BackgroundColor3 = Theme.Violet,
			}, Fast)
		end
	end

	CurrentPage = name
end

for name, data in pairs(PageButtons) do
	data.Button.Activated:Connect(function()
		ShowPage(name)
	end)

	data.Button.MouseEnter:Connect(function()
		if CurrentPage ~= name then
			Tween(data.Button, {
				BackgroundTransparency = 0.35,
				BackgroundColor3 = Color3.fromRGB(36, 28, 58),
			}, Fast)
			Tween(data.Icon, {TextColor3 = Theme.SoftWhite}, Fast)
		end
	end)

	data.Button.MouseLeave:Connect(function()
		if CurrentPage ~= name then
			Tween(data.Button, {
				BackgroundTransparency = 1,
				BackgroundColor3 = Color3.fromRGB(38, 28, 64),
			}, Fast)
			Tween(data.Icon, {TextColor3 = Theme.Gray}, Fast)
		end
	end)
end

local function CreateToggle(page, name, description, getter, setter)
	local row = Card(page, 50)

	local title = New("TextLabel", {
		Position = UDim2.fromOffset(12, 6),
		Size = UDim2.new(1, -75, 0, 18),

		BackgroundTransparency = 1,

		Text = name,
		TextColor3 = Theme.White,
		TextSize = 12,

		Font = Enum.Font.GothamSemibold,

		TextXAlignment = Enum.TextXAlignment.Left,
	}, row)

	local desc = New("TextLabel", {
		Position = UDim2.fromOffset(12, 25),
		Size = UDim2.new(1, -75, 0, 14),

		BackgroundTransparency = 1,

		Text = description,
		TextColor3 = Theme.Gray,
		TextSize = 8,

		Font = Enum.Font.Gotham,

		TextXAlignment = Enum.TextXAlignment.Left,
	}, row)

	local switch = New("TextButton", {
		AnchorPoint = Vector2.new(1, 0.5),
		Position = UDim2.new(1, -11, 0.5, 0),

		Size = UDim2.fromOffset(44, 23),

		BackgroundColor3 = Theme.Off,
		BorderSizePixel = 0,

		Text = "",
		AutoButtonColor = false,
	}, row)

	Corner(switch, 100)

	local knob = New("Frame", {
		Position = UDim2.fromOffset(3, 3),
		Size = UDim2.fromOffset(17, 17),

		BackgroundColor3 = Color3.fromRGB(190, 187, 205),
		BorderSizePixel = 0,
	}, switch)

	Corner(knob, 100)

	local function UpdateVisual(on)
		if on then
			Tween(
				switch,
				{BackgroundColor3 = Theme.Violet},
				Fast
			)

			Tween(
				knob,
				{
					Position = UDim2.new(1, -20, 0, 3),
					BackgroundColor3 = Theme.White
				},
				Fast
			)

			Tween(
				row,
				{BackgroundColor3 = Color3.fromRGB(29, 23, 49)},
				Fast
			)
		else
			Tween(
				switch,
				{BackgroundColor3 = Theme.Off},
				Fast
			)

			Tween(
				knob,
				{
					Position = UDim2.fromOffset(3, 3),
					BackgroundColor3 = Color3.fromRGB(190, 187, 205)
				},
				Fast
			)

			Tween(
				row,
				{BackgroundColor3 = Theme.Card},
				Fast
			)
		end
	end

	UpdateVisual(getter())

	switch.Activated:Connect(function()
		local newState = not getter()

		setter(newState)
		UpdateVisual(newState)
	end)

	return {
		SetVisual = UpdateVisual
	}
end

local function MakeDraggable(object, handle)
	local dragging = false
	local dragStart
	local startPosition
	local dragInput

	handle.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then

			dragging = true
			dragStart = input.Position
			startPosition = object.Position
			dragInput = input
		end
	end)

	handle.InputChanged:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseMovement
			or input.UserInputType == Enum.UserInputType.Touch then

			dragInput = input
		end
	end)

	UserInputService.InputChanged:Connect(function(input)
		if dragging and input == dragInput then

			local delta =
				input.Position - dragStart

			object.Position = UDim2.new(
				startPosition.X.Scale,
				startPosition.X.Offset + delta.X,

				startPosition.Y.Scale,
				startPosition.Y.Offset + delta.Y
			)
		end
	end)

	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then

			dragging = false
		end
	end)
end

--[[
	When you fly very high, Roblox streams the map out → lag spikes + enemies catch up.
	These controls force higher quality and try to keep more of the world loaded.
]]

local RenderState = {
	MaxQuality = false,
	HighStream = false,
}

local SavedRender = {
	QualityLevel = nil,
	MeshPartDetailLevel = nil,
	EagerBulkExecution = nil,
	StreamingMinRadius = nil,
	StreamingTargetRadius = nil,
}

local function ApplyMaxQuality(on)
	local ok, err = pcall(function()
		local s = settings()
		local rendering = s.Rendering

		if on then
			if SavedRender.QualityLevel == nil then
				SavedRender.QualityLevel = rendering.QualityLevel
			end
			-- Highest fixed quality
			rendering.QualityLevel = Enum.QualityLevel.Level21

			pcall(function()
				rendering.MeshPartDetailLevel = Enum.MeshPartDetailLevel.Level04
			end)
			pcall(function()
				rendering.EagerBulkExecution = true
			end)
			pcall(function()
				rendering.EditQualityLevel = Enum.QualityLevel.Level21
			end)
		else
			if SavedRender.QualityLevel ~= nil then
				rendering.QualityLevel = SavedRender.QualityLevel
			else
				rendering.QualityLevel = Enum.QualityLevel.Automatic
			end
		end
	end)

	if not ok and status then
		status.Text = "Max Quality failed: " .. tostring(err)
	end
end

CreateToggle(
	ToolsPage,
	"Max Quality",
	"Forces highest graphics (Level 21). Better map detail when high up.",
	function()
		return RenderState.MaxQuality
	end,
	function(on)
		RenderState.MaxQuality = on
		ApplyMaxQuality(on)
		status.Text = on and "Max Quality ON (Level 21)." or "Max Quality OFF."
	end
)

local function ApplyStreamingRadius(radius)
	radius = math.clamp(tonumber(radius) or 2048, 64, 16384)

	local ok, err = pcall(function()
		local ws = workspace

		-- Save originals once
		if SavedRender.StreamingMinRadius == nil then
			pcall(function()
				SavedRender.StreamingMinRadius = ws.StreamingMinRadius
				SavedRender.StreamingTargetRadius = ws.StreamingTargetRadius
			end)
		end

		-- These only work if the place has StreamingEnabled
		pcall(function()
			ws.StreamingMinRadius = math.min(radius, 1024)
		end)
		pcall(function()
			ws.StreamingTargetRadius = radius
		end)
	end)

	if status then
		if ok then
			status.Text = "Streaming radius set to " .. tostring(radius) .. " studs."
		else
			status.Text = "Streaming locked by game (use Max Quality)."
		end
	end

	return ok
end

	ApplyStreamingRadius(StreamBox.Text)
end)

-- Optional low-quality FPS mode (kept, but not recommended for high flying)
local FPSBoostState = false
local SavedQuality

local function SetFPSBoost(value)
	FPSBoostState = value

	local s = settings()

	if value then
		-- Don't fight Max Quality
		if RenderState.MaxQuality then
			RenderState.MaxQuality = false
		end

		SavedQuality = s.Rendering.QualityLevel
		s.Rendering.QualityLevel = Enum.QualityLevel.Level01
		status.Text = "FPS Boost ON (not ideal while flying high)."
	else
		if SavedQuality then
			s.Rendering.QualityLevel = SavedQuality
		end
		status.Text = "FPS Boost OFF."
	end
end
